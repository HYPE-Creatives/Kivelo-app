================================================================================
KIVELO APP - CHAT & CONVERSATION INTEGRATION GUIDE
================================================================================

This document provides frontend integration guidance for the Chat & Conversation
features in the Kivelo Family Wellness app. It covers:
- AI Chat (Child's AI Counselor)
- Family Chat (Parent-Child, Sibling messaging)
- Safety Monitoring & Parent Oversight
- Message Management

================================================================================
                        📋 FEATURE OVERVIEW
================================================================================

The conversation system supports THREE types of chats:

1. AI CHAT (ai_chat)
   - Child-only feature
   - AI counselor with mood-aware responses
   - Loads mood context, journal themes, activity engagement
   - Session-based with stats tracking

2. FAMILY CHAT (family_chat)
   - Multi-participant (parents + children)
   - General family discussions

3. DIRECT CHATS (parent_child, sibling)
   - 1-on-1 conversations
   - parent_child: Between a parent and their child
   - sibling: Between siblings

SAFETY FEATURES:
- All messages are checked for safety concerns
- Flagged messages trigger parent notifications
- Parents can review flagged conversations

================================================================================
                        🔗 API BASE URL
================================================================================

All conversation endpoints: /api/v1/conversations/...

Headers Required:
- Content-Type: application/json
- Authorization: Bearer <jwt_token>

================================================================================
                        📝 TYPESCRIPT TYPES
================================================================================

```typescript
// Message types
interface Message {
  _id: string;
  sender?: string;                    // User ID (null for AI messages)
  role: 'user' | 'assistant' | 'system' | 'parent' | 'child';
  content: string;
  type: 'text' | 'suggestion' | 'prompt' | 'activity' | 'mood_insight' | 'safety_alert' | 'image' | 'voice';
  metadata?: {
    moodScore?: number;
    trustZone?: string;
    aiConfidence?: number;
    suggestedActivity?: string;
    readBy?: { user: string; readAt: string }[];
  };
  flagged?: boolean;
  flagReason?: string;
  createdAt: string;
  edited?: boolean;
  editedAt?: string;
}

// Conversation types
interface Conversation {
  _id: string;
  type: 'ai_chat' | 'family_chat' | 'parent_child' | 'sibling';
  childId?: string;                   // For AI chats
  participants?: Participant[];        // For family chats
  familyId?: string;
  context?: MoodContext;              // AI chat mood context
  messages: Message[];
  moodAtStart?: {
    emoji: string;
    moodScore: number;
    trustZone: string;
  };
  isActive: boolean;
  safetyFlags?: SafetyFlags;
  stats?: ConversationStats;
  createdAt: string;
  updatedAt: string;
}

interface Participant {
  user: {
    _id: string;
    name: string;
    email?: string;
    avatar?: { url: string };
    role: 'parent' | 'child';
  };
  role: 'parent' | 'child';
  joinedAt: string;
  lastReadAt?: string;
}

interface MoodContext {
  currentMood?: {
    emoji: string;
    moodScore: number;
    trustZone: 'green' | 'yellow' | 'orange' | 'red';
    lastCheckinAt: string;
  };
  moodTrend?: {
    weeklyAverage: number | null;
    trend: 'improving' | 'stable' | 'declining';
    daysTracked: number;
  };
  recentJournalThemes?: string[];
  interests?: string[];
  activityEngagement?: {
    completedThisWeek: number;
    favoriteCategory: string | null;
  };
}

interface SafetyFlags {
  hasConcerningContent: boolean;
  flagCount: number;
  lastFlaggedAt?: string;
  parentNotified: boolean;
  requiresReview: boolean;
}

interface ConversationStats {
  totalMessages: number;
  userMessages: number;
  aiMessages: number;
  averageResponseTime?: number;
  sessionDuration?: number;           // In seconds
}

// Family conversation list item
interface ConversationListItem {
  _id: string;
  type: 'family_chat' | 'parent_child' | 'sibling';
  participants: Participant[];
  lastMessage?: {
    content: string;                  // First 50 chars
    createdAt: string;
    senderRole: string;
  };
  unreadCount: number;
  updatedAt: string;
}
```

1. GET OR CREATE AI CONVERSATION
--------------------------------------------------------------------------------
Endpoint: GET /api/v1/conversations/ai
Protected: Yes (Child role required)
Headers: Authorization: Bearer <child_token>

Description: Gets the active AI conversation or creates a new one.
Automatically loads the child's mood context for personalized responses.

Success Response (200):
{
  "success": true,
  "data": {
    "conversationId": "conv_id",
    "type": "ai_chat",
    "context": {
      "currentMood": {
        "emoji": "😊",
        "moodScore": 8,
        "trustZone": "green",
        "lastCheckinAt": "2024-01-01T10:00:00.000Z"
      },
      "moodTrend": {
        "weeklyAverage": 7.5,
        "trend": "improving",
        "daysTracked": 5
      },
      "recentJournalThemes": ["school", "friends"],
      "interests": ["gaming", "reading"],
      "activityEngagement": {
        "completedThisWeek": 3,
        "favoriteCategory": "education"
      }
    },
    "messages": [ /* last 20 messages */ ],
    "moodAtStart": {
      "emoji": "😊",
      "moodScore": 8,
      "trustZone": "green"
    },
    "stats": {
      "totalMessages": 15,
      "userMessages": 8,
      "aiMessages": 7
    }
  }
}

Frontend Implementation:
```typescript
const getOrCreateAIConversation = async () => {
  try {
    const token = await AsyncStorage.getItem('kivelo_access_token');
    if (!token) throw new Error('Not authenticated');

    const response = await fetch(`${API_BASE_URL}/conversations/ai`, {
      method: 'GET',
      headers: {
        'Authorization': `Bearer ${token}`,
        'Content-Type': 'application/json'
      }
    });

    const data = await response.json();
    
    if (data.success) {
      return {
        conversationId: data.data.conversationId,
        messages: data.data.messages,
        context: data.data.context,
        moodAtStart: data.data.moodAtStart
      };
    }
    throw new Error(data.message);
  } catch (error) {
    console.error('Error getting AI conversation:', error);
    throw error;
  }
};
```

```typescript
const sendMessage = async (conversationId: string, content: string, type: string = 'text') => {

      return {
        message: data.data.message,
        safetyCheck: data.data.safetyCheck

    throw new Error(data.message);
  } catch (error) {
    console.error('Error sending message:', error);
    throw error;
  }
};
```

4. END AI CONVERSATION
--------------------------------------------------------------------------------
Endpoint: POST /api/v1/conversations/ai/:conversationId/end
Protected: Yes (Child role required)
Headers: Authorization: Bearer <child_token>

Description: Ends the current AI conversation. Next call to GET /ai creates new one.

Success Response (200):
{
  "success": true,
  "message": "Conversation ended",
  "data": {
    "sessionDuration": 300,           // seconds
    "totalMessages": 15
  }
}

Frontend Implementation:
```typescript
const endAIConversation = async (conversationId: string) => {
  try {
    const token = await AsyncStorage.getItem('kivelo_access_token');

      method: 'POST',
      headers: {
        'Authorization': `Bearer ${token}`,
        'Content-Type': 'application/json'
      }
    });

    const data = await response.json();
    return { success: data.success, sessionStats: data.data };
```

3. GET CONVERSATION HISTORY
--------------------------------------------------------------------------------
Endpoint: GET /api/v1/conversations/:conversationId
Protected: Yes
Headers: Authorization: Bearer <token>

Query Parameters:
- limit: number (default: 50) - Number of messages to return
- before: ISO date string - Get messages before this timestamp (for pagination)

Success Response (200):
{
  "success": true,
  "data": {
    "conversationId": "conv_id",
    "type": "parent_child",
    "messages": [
      {
        "_id": "msg_id",
        "sender": {
          "_id": "user_id",
          "name": "Mom",
          "avatar": { "url": "https://..." }
        },
        "role": "parent",
        "content": "How was school today?",
        "type": "text",
        "createdAt": "2024-01-01T15:00:00.000Z"
      }
    ],
    "hasMore": true,
    "stats": {
      "totalMessages": 150,
      "userMessages": 75,
      "aiMessages": 0
    }
  }
}

Frontend Implementation:
```typescript
const getConversationHistory = async (
  conversationId: string, 
  limit: number = 50, 
  before?: string
) => {
  try {
    const token = await AsyncStorage.getItem('kivelo_access_token');
    
    let url = `${API_BASE_URL}/conversations/${conversationId}?limit=${limit}`;
    if (before) {
      url += `&before=${encodeURIComponent(before)}`;
    }

    const response = await fetch(url, {
      method: 'GET',
      headers: {
        'Authorization': `Bearer ${token}`,
        'Content-Type': 'application/json'
      }
    });

    const data = await response.json();
    
    if (data.success) {
      return {
        messages: data.data.messages,
        hasMore: data.data.hasMore,
        stats: data.data.stats
      };
    }
    throw new Error(data.message);
  } catch (error) {
    console.error('Error getting conversation history:', error);
    throw error;
  }
};
```

================================================================================
                        📦 MESSAGE ARCHIVING (Hybrid Storage)
================================================================================

Endpoint: POST /api/v1/conversations/:conversationId/archive
Protected: Yes
Headers: Authorization: Bearer <token>

Description: Archives old messages from server and returns them for local storage.
Keeps recent messages in database for quick access.

Request Body:
{
  "keepCount": 50    // Number of recent messages to keep in DB
}

Success Response (200):
{
  "success": true,
  "data": {
    "archivedCount": 100,
    "archivedMessages": [ /* array of archived messages to store locally */ ],
    "remainingInDB": 50
  }
}

Frontend Pattern (Hybrid Storage):
```typescript
const archiveMessages = async (conversationId: string, keepCount: number = 50) => {
  try {
    const token = await AsyncStorage.getItem('kivelo_access_token');
    
    const response = await fetch(`${API_BASE_URL}/conversations/${conversationId}/archive`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${token}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ keepCount })
    });

    const data = await response.json();
    
    if (data.success && data.data.archivedMessages.length > 0) {
      // Store archived messages locally
      const localKey = `archived_messages_${conversationId}`;
      const existingArchive = await AsyncStorage.getItem(localKey);
      const existing = existingArchive ? JSON.parse(existingArchive) : [];
      
      // Append new archived messages
      const updated = [...existing, ...data.data.archivedMessages];
      await AsyncStorage.setItem(localKey, JSON.stringify(updated));
      
      return {
        archivedCount: data.data.archivedCount,
        totalLocal: updated.length
      };
    }
    return { archivedCount: 0, totalLocal: 0 };
  } catch (error) {
    console.error('Error archiving messages:', error);
    throw error;
  }
};
```

AI CHAT FLOW (Child):
--------------------------------------------------------------------------------
1. Child opens AI Helper screen
2. Call GET /conversations/ai to get/create conversation
3. Display mood context in UI (show child's current mood)
4. Show existing messages (from response)
5. Child sends message → POST /conversations/message
6. Get AI response from your AI service
7. Save AI response → POST /conversations/ai/response
8. Display AI message in chat
9. (Optional) Child ends conversation → POST /conversations/ai/{id}/end

FAMILY CHAT FLOW:
--------------------------------------------------------------------------------
1. User opens Chat screen
2. Call GET /conversations/family to list conversations
3. Display conversation list with unread counts
4. User taps conversation → GET /conversations/{id} for history
5. Call POST /conversations/{id}/read to mark read
6. User sends message → POST /conversations/message
7. Display message (optimistic UI)
8. (If new chat) → POST /conversations/family with participantIds

PARENT OVERSIGHT FLOW:
--------------------------------------------------------------------------------
1. Parent sees notification badge for flagged content
2. Parent opens oversight screen
3. Call GET /conversations/parent/flagged
4. Display flagged conversations with excerpts
5. Parent reviews conversation
6. Call POST /conversations/parent/review/{id} with notes
7. (Optional) View insights → GET /conversations/parent/child/{childId}/insights

================================================================================
                        ⚠️ SAFETY MONITORING
================================================================================

All messages are automatically checked for safety concerns.
The system checks for:
- Concerning language or topics
- Potential signs of distress
- Inappropriate content

When a message is flagged:
1. Message is saved with flagged: true
2. Conversation's safetyFlags are updated
3. High severity triggers IMMEDIATE parent notification
4. Conversation marked as requiresReview: true

Safety Check Response in Send Message:
```typescript
{
  "safetyCheck": {
    "flagged": true,
    "category": "potential_concern"  // or other category
  }
}
```

Frontend Handling:
```typescript
const result = await sendMessage(conversationId, content);

if (result.safetyCheck?.flagged) {
  // Message was flagged but sent
  // Optionally show gentle reminder to child
  console.log('Message flagged:', result.safetyCheck.category);
}
```

```typescript
// context/ConversationContext.tsx
import React, { createContext, useContext, useState, useCallback, ReactNode } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';

const API_BASE_URL = 'https://family-wellness.onrender.com/api/v1';
const ACCESS_TOKEN_KEY = 'kivelo_access_token';

interface ConversationContextType {
  // AI Chat
  aiConversation: Conversation | null;
  getAIConversation: () => Promise<void>;
  sendAIMessage: (content: string) => Promise<Message>;
  addAIResponse: (content: string, metadata?: any) => Promise<Message>;
  endAIConversation: () => Promise<void>;
  
  // Family Chat
  familyConversations: ConversationListItem[];
  getFamilyConversations: () => Promise<void>;
  createConversation: (participantIds: string[], type?: string) => Promise<Conversation>;
  getConversationHistory: (conversationId: string, limit?: number) => Promise<Message[]>;
  sendMessage: (conversationId: string, content: string) => Promise<Message>;
  markAsRead: (conversationId: string) => Promise<void>;
  
  // Message management
  editMessage: (conversationId: string, messageId: string, content: string) => Promise<void>;
  deleteMessage: (conversationId: string, messageId: string) => Promise<void>;
  
  // Parent oversight
  flaggedConversations: FlaggedConversation[];
  getFlaggedConversations: () => Promise<void>;
  reviewConversation: (conversationId: string, notes: string, acknowledged: boolean) => Promise<void>;
  getChildInsights: (childId: string) => Promise<ChildInsights>;
  
  loading: boolean;
  error: string | null;
}

export const ConversationProvider = ({ children }: { children: ReactNode }) => {
  const [aiConversation, setAIConversation] = useState(null);
  const [familyConversations, setFamilyConversations] = useState([]);
  const [flaggedConversations, setFlaggedConversations] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  // Helper for authenticated requests
  const authFetch = async (endpoint: string, options: RequestInit = {}) => {
    const token = await AsyncStorage.getItem(ACCESS_TOKEN_KEY);
    if (!token) throw new Error('Not authenticated');

    const response = await fetch(`${API_BASE_URL}${endpoint}`, {
      ...options,
      headers: {
        'Authorization': `Bearer ${token}`,
        'Content-Type': 'application/json',
        ...options.headers,
      },
    });

    const data = await response.json();
    if (!response.ok) throw new Error(data.message || 'Request failed');
    return data;
  };

  const getAIConversation = useCallback(async () => {
    setLoading(true);
    try {
      const data = await authFetch('/conversations/ai');
      setAIConversation(data.data);
    } catch (err: any) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  }, []);

  const sendMessage = useCallback(async (conversationId: string, content: string) => {
    const data = await authFetch('/conversations/message', {
      method: 'POST',
      body: JSON.stringify({ conversationId, content, type: 'text' }),
    });
    return data.data.message;
  }, []);

  // ... implement other methods similarly

  return (
    <ConversationContext.Provider value={{
      aiConversation,
      getAIConversation,
      // ... other values
      loading,
      error,
    }}>
      {children}
    </ConversationContext.Provider>
  );
};
```
